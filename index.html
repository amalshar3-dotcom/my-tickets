<script type="module">
import { initializeApp } from
"https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
import {
getAuth, onAuthStateChanged, createUserWithEmailAndPassword,
signInWithEmailAndPassword, sendEmailVerification,
sendPasswordResetEmail, signOut
} from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js";
import {
getFirestore, collection, addDoc, query, orderBy, limit, getDocs
} from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js";
/* ============================================================
ВСТАВЬ СЮДА СВОЙ firebaseConfig (из консоли Firebase).
Меняй только значения в кавычках. Остальное не трогай.
============================================================ */
const firebaseConfig = {
apiKey: "ВСТАВЬ_СЮДА",
authDomain: "ВСТАВЬ_СЮДА",
projectId: "ВСТАВЬ_СЮДА",
storageBucket: "ВСТАВЬ_СЮДА",
messagingSenderId: "ВСТАВЬ_СЮДА",
appId: "ВСТАВЬ_СЮДА"
};
/* ============================================================ */
const PRICE = 48;
const TICKET_MS = (59 * 60 + 59) * 1000;
const CARRIERS = {
"Автобус": "Автобусное депо «Северное»",
"Троллейбус": "Троллейбусное депо «Речное»",
"Трамвай": "Трамвайное депо «Лесное»"
};
const $ = id => document.getElementById(id);
const pages = { auth: $('pageAuth'), form: $('pageForm'), ticket: $('pageTicket')
};
function show(name) {
Object.entries(pages).forEach(([k, el]) => el.classList.toggle('hidden', k !==
name));
window.scrollTo(0, 0);
}
function setMsg(el, text) {
el.textContent = text || '';
el.classList.toggle('hidden', !text);
}
/* ---------- Firebase ---------- */
const configReady = !String(firebaseConfig.apiKey).includes('ВСТАВЬ');
let auth = null, db = null, currentUser = null;
if (configReady) {
const app = initializeApp(firebaseConfig);
auth = getAuth(app);
db = getFirestore(app);
}
/* ---------- Хранилище на устройстве (черновик и последний экран) ---------- */
const store = {
get(k) { try { return JSON.parse(localStorage.getItem(k)); } catch { return
null; } },
set(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); } catch {} },
del(k) { try { localStorage.removeItem(k); } catch {} }
};
/* ---------- Регистрация / вход ---------- */
let mode = 'register';
function setMode(m) {
mode = m;
const reg = m === 'register';
$('tabRegister').classList.toggle('on', reg);
$('tabLogin').classList.toggle('on', !reg);
$('authTitle').textContent = reg ? 'Создай аккаунт' : 'Вход в аккаунт';
$('authLead').textContent = reg
? 'Укажи почту и придумай пароль. Билеты будут сохраняться в твоём аккаунте.'
: 'Введи почту и пароль, с которыми регистрировалась(ся).';
$('authSubmit').textContent = reg ? 'Зарегистрироваться' : 'Войти';
$('password').autocomplete = reg ? 'new-password' : 'current-password';
$('resetBtn').classList.toggle('hidden', reg);
setMsg($('authError'), ''); setMsg($('authInfo'), '');
}
$('tabRegister').addEventListener('click', () => setMode('register'));
$('tabLogin').addEventListener('click', () => setMode('login'));
function authErrorText(e) {
const map = {
'auth/email-already-in-use': 'Эта почта уже зарегистрирована. Перейди на
вкладку «Вход».',
'auth/invalid-email': 'Проверь почту: она написана неверно.',
'auth/weak-password': 'Пароль слишком простой. Нужно минимум 6 символов.',
'auth/invalid-credential': 'Неверная почта или пароль.',
'auth/user-not-found': 'Такого аккаунта нет. Перейди на вкладку
«Регистрация».',
'auth/wrong-password': 'Неверный пароль.',
'auth/too-many-requests': 'Слишком много попыток. Подожди несколько минут.',
'auth/network-request-failed': 'Нет интернета. Проверь соединение.',
'auth/unauthorized-domain': 'Этот сайт не добавлен в Authorized domains в
Firebase.'
};
return map[e && e.code] || 'Что-то пошло не так. Попробуй ещё раз.';
}
$('authForm').addEventListener('submit', async ev => {
ev.preventDefault();
setMsg($('authError'), ''); setMsg($('authInfo'), '');
if (!configReady) {
setMsg($('authError'), 'Firebase не подключён. Вставь firebaseConfig в
index.html (см. инструкцию).');
return;
}
const email = $('email').value.trim();
const pass = $('password').value;
const btn = $('authSubmit');
btn.disabled = true;
try {
if (mode === 'register') {
const cred = await createUserWithEmailAndPassword(auth, email, pass);
await sendEmailVerification(cred.user);
} else {
await signInWithEmailAndPassword(auth, email, pass);
}
// дальше сработает onAuthStateChanged
} catch (e) {
setMsg($('authError'), authErrorText(e));
} finally {
btn.disabled = false;
}
});
$('resetBtn').addEventListener('click', async () => {
setMsg($('authError'), ''); setMsg($('authInfo'), '');
const email = $('email').value.trim();
if (!email) { setMsg($('authError'), 'Сначала впиши почту в поле выше.');
return; }
try {
await sendPasswordResetEmail(auth, email);
setMsg($('authInfo'), 'Письмо для смены пароля отправлено. Проверь почту и
«Спам».');
} catch (e) { setMsg($('authError'), authErrorText(e)); }
});
/* ---------- Подтверждение почты ---------- */
$('verifiedBtn').addEventListener('click', async () => {
setMsg($('verifyError'), ''); setMsg($('verifyInfo'), '');
try {
await auth.currentUser.reload();
if (auth.currentUser.emailVerified) {
await enterApp(auth.currentUser);
} else {
setMsg($('verifyError'), 'Почта пока не подтверждена. Нажми на ссылку в
письме и попробуй снова.');
}
} catch (e) { setMsg($('verifyError'), authErrorText(e)); }
});
$('resendBtn').addEventListener('click', async () => {
setMsg($('verifyError'), ''); setMsg($('verifyInfo'), '');
try {
await sendEmailVerification(auth.currentUser);
setMsg($('verifyInfo'), 'Письмо отправлено ещё раз.');
} catch (e) { setMsg($('verifyError'), authErrorText(e)); }
});
$('verifyLogout').addEventListener('click', doLogout);
$('logoutBtn').addEventListener('click', doLogout);
async function doLogout() {
store.del('view'); store.del('ticketId');
if (auth) await signOut(auth);
}
/* ---------- Состояние входа (сохраняется при повторном открытии) ---------- */
if (configReady) {
onAuthStateChanged(auth, async user => {
currentUser = user;
if (!user) {
$('authBox').classList.remove('hidden');
$('verifyBox').classList.add('hidden');
show('auth');
return;
}
if (!user.emailVerified) {
$('authBox').classList.add('hidden');
$('verifyBox').classList.remove('hidden');
$('verifyEmail').textContent = user.email;
show('auth');
return;
}
await enterApp(user);
});
} else {
setMode('register');
show('auth');
}
/* ---------- Вход в приложение ---------- */
let history = [];
async function enterApp(user) {
currentUser = user;
restoreDraft();
show('form');
await loadHistory();
// Если билет был открыт перед закрытием сайта — открываем его снова
const lastId = store.get('ticketId');
if (store.get('view') === 'ticket' && lastId) {
const t = history.find(x => x.id === lastId);
if (t) openTicket(t);
}
}
/* ---------- Черновик формы ---------- */
const numberInput = $('transportNumber');
const qtyInput = $('quantity');
const typeBoxes = () =>
Array.from(document.querySelectorAll('input[name="transportType"]'));
function saveDraft() {
store.set('draft', {
number: numberInput.value,
qty: qtyInput.value,
types: typeBoxes().filter(b => b.checked).map(b => b.value)
});
}
function restoreDraft() {
const d = store.get('draft');
if (!d) return;
numberInput.value = d.number || '';
qtyInput.value = d.qty || 1;
typeBoxes().forEach(b => { b.checked = (d.types || []).includes(b.value); });
}
numberInput.addEventListener('input', saveDraft);
qtyInput.addEventListener('input', saveDraft);
typeBoxes().forEach(b => b.addEventListener('change', saveDraft));
/* ---------- Создание билета ---------- */
function getTicketWord(n) {
const lastTwo = n % 100;
if (lastTwo >= 11 && lastTwo <= 14) return 'билетов';
const last = n % 10;
if (last === 1) return 'билет';
if (last >= 2 && last <= 4) return 'билета';
return 'билетов';
}
function randomDigits() { return String(Math.floor(1000 + Math.random() * 9000));
}
$('ticketForm').addEventListener('submit', async ev => {
ev.preventDefault();
const number = numberInput.value.trim();
const quantity = Number(qtyInput.value);
const types = typeBoxes().filter(b => b.checked).map(b => b.value);
const err = $('errorMessage');
setMsg(err, '');
if (!number) return setMsg(err, 'Введи номер транспорта.');
if (types.length === 0) return setMsg(err, 'Выбери автобус, троллейбус или
трамвай.');
if (!Number.isInteger(quantity) || quantity < 1 || quantity > 99)
return setMsg(err, 'Количество билетов должно быть от 1 до 99.');
const digits = randomDigits();
const now = Date.now();
const ticket = {
number, types, quantity,
title: types.map(t => t + ' ' + number).join(' / '),
carrier: Array.from(new Set(types.map(t => CARRIERS[t]))).join(' / '),
regNumber: 'ИГРА-' + digits,
boardNumber: '№' + digits,
total: quantity * PRICE,
createdAt: now,
endTime: now + TICKET_MS
};
const btn = $('ticketSubmit');
btn.disabled = true;
try {
const ref = await addDoc(collection(db, 'users', currentUser.uid, 'tickets'),
ticket);
ticket.id = ref.id;
} catch (e) {
ticket.id = 'local-' + now; // билет всё равно покажем
console.error(e);
}
btn.disabled = false;
history.unshift(ticket);
renderHistory();
openTicket(ticket);
});
/* ---------- История ---------- */
async function loadHistory() {
try {
const q = query(
collection(db, 'users', currentUser.uid, 'tickets'),
orderBy('createdAt', 'desc'), limit(30)
);
const snap = await getDocs(q);
history = snap.docs.map(d => ({ id: d.id, ...d.data() }));
} catch (e) {
console.error(e);
history = [];
}
renderHistory();
}
function fmtDateTime(ms) {
const d = new Date(ms);
return new Intl.DateTimeFormat('ru-RU', {
day: 'numeric', month: 'long', hour: '2-digit', minute: '2-digit'
}).format(d);
}
function renderHistory() {
const box = $('historyList');
box.innerHTML = '';
if (!history.length) {
box.innerHTML = '<div class="empty">Пока нет билетов. Оформи первый выше.
</div>';
return;
}
history.forEach(t => {
const live = t.endTime > Date.now();
const btn = document.createElement('button');
btn.type = 'button';
btn.className = 'h-item';
btn.innerHTML =
'<div class="t"><b></b><small></small></div><span class="badge"></span>';
btn.querySelector('b').textContent = t.title;
btn.querySelector('small').textContent =
t.quantity + ' ' + getTicketWord(t.quantity) + ' · ' +
fmtDateTime(t.createdAt);
const badge = btn.querySelector('.badge');
badge.textContent = live ? 'Действует' : 'Истёк';
if (live) badge.classList.add('live');
btn.addEventListener('click', () => openTicket(t));
box.appendChild(btn);
});
}
/* ---------- Показ билета ---------- */
let timerInterval = null;
let endTime = 0;
function openTicket(t) {
$('transportTitle').textContent = t.title;
$('ticketCount').textContent = t.quantity;
$('ticketCountTop').textContent = t.quantity + ' ' + getTicketWord(t.quantity) +
' · игра';
$('totalPrice').textContent = (t.quantity * PRICE).toLocaleString('ru-RU') + '
₽';
$('carrier').textContent = t.carrier;
$('regNumber').textContent = t.regNumber;
$('boardNumber').textContent = t.boardNumber;
endTime = t.endTime;
const expiry = new Date(endTime);
const dateText = new Intl.DateTimeFormat('ru-RU', {
day: 'numeric', month: 'long', year: 'numeric'
}).format(expiry);
$('expiryDate').textContent =
'до ' + dateText + ' г. до ' + pad(expiry.getHours()) + ':' +
pad(expiry.getMinutes());
store.set('view', 'ticket');
store.set('ticketId', t.id);
show('ticket');
if (timerInterval) clearInterval(timerInterval);
updateTimer();
timerInterval = setInterval(updateTimer, 50);
}
$('backBtn').addEventListener('click', () => {
if (timerInterval) { clearInterval(timerInterval); timerInterval = null; }
store.set('view', 'form');
renderHistory();
show('form');
});
function pad(n) { return String(n).padStart(2, '0'); }
function updateTimer() {
const remaining = Math.max(0, endTime - Date.now());
const totalSeconds = Math.floor(remaining / 1000);
const min = pad(Math.floor(totalSeconds / 60));
const sec = pad(totalSeconds % 60);
const ms = pad(Math.floor((remaining % 1000) / 10));
$('min1').textContent = min[0]; $('min2').textContent = min[1];
$('sec1').textContent = sec[0]; $('sec2').textContent = sec[1];
$('ms1').textContent = ms[0]; $('ms2').textContent = ms[1];
$('timerTitle').textContent = remaining > 0 ? 'Игровой таймер' : 'Время вышло';
if (remaining <= 0 && timerInterval) {
clearInterval(timerInterval);
timerInterval = null;
}
}
